### FILE: LibrarySystem.java
```java
public class LibrarySystem {
    public static void main(String[] args) {
        // Create the table module with different book types
        TableModule<Book> bookTable = new TableModule<>();
        
        // Add books to the system
        Book book1 = new Book("1234567890", "Java Programming", "John Doe");
        Book book2 = new Book("0987654321", "Design Patterns", "Gang of Four");
        
        bookTable.add(book1);
        bookTable.add(book2);
        
        // Display all books
        System.out.println("All Books:");
        for (Book book : bookTable.getAll()) {
            System.out.println("- " + book.getTitle() + " by " + book.getAuthor());
        }
    }
}
```

### FILE: TableModule.java
```java
import java.util.*;

public class TableModule<T> {
    private List<T> items;
    
    public TableModule() {
        this.items = new ArrayList<>();
    }
    
    public void add(T item) {
        items.add(item);
    }
    
    public List<T> getAll() {
        return new ArrayList<>(items);
    }
    
    public T get(int index) {
        return items.get(index);
    }
    
    public int size() {
        return items.size();
    }
}
```

### FILE: Book.java
```java
public class Book {
    private String isbn;
    private String title;
    private String author;
    
    public Book(String isbn, String title, String author) {
        this.isbn = isbn;
        this.title = title;
        this.author = author;
    }
    
    public String getIsbn() {
        return isbn;
    }
    
    public String getTitle() {
        return title;
    }
    
    public String getAuthor() {
        return author;
    }
}
```
