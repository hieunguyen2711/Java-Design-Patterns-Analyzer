### FILE: LibraryItem.java
```java
package library;

public abstract class LibraryItem {
    protected String title;
    protected String itemId;
    protected boolean isAvailable;
    
    public LibraryItem(String title, String itemId) {
        this.title = title;
        this.itemId = itemId;
        this.isAvailable = true;
    }
    
    public abstract double calculateLateFee(int daysOverdue);
    
    public String getTitle() {
        return title;
    }
    
    public String getItemId() {
        return itemId;
    }
    
    public boolean isAvailable() {
        return isAvailable;
    }
    
    public void setAvailable(boolean available) {
        isAvailable = available;
    }
}
```

### FILE: Book.java
```java
package library;

public class Book extends LibraryItem {
    private String author;
    private int pages;
    
    public Book(String title, String itemId, String author, int pages) {
        super(title, itemId);
        this.author = author;
        this.pages = pages;
    }
    
    @Override
    public double calculateLateFee(int daysOverdue) {
        return daysOverdue * 0.50; // $0.50 per day for books
    }
    
    public String getAuthor() {
        return author;
    }
    
    public int getPages() {
        return pages;
    }
}
```

### FILE: Magazine.java

```java
package library;

public class Magazine extends LibraryItem {
    private String publisher;
    private int issueNumber;
    
    public Magazine(String title, String itemId, String publisher, int issueNumber) {
        super(title, itemId);
        this.publisher = publisher;
        this.issueNumber = issueNumber;
    }
    
    @Override
    public double calculateLateFee(int daysOverdue) {
        return daysOverdue * 1.00; // $1.00 per day for magazines
    }
    
    public String getPublisher() {
        return publisher;
    }
    
    public int getIssueNumber() {
        return issueNumber;
    }
}
```
