    private List<Member> members;
    
    public LibraryMediator() {
        this.books = new ArrayList<>();
        this.members = new ArrayList<>();
    }

    public void registerBook(Book book) {
        books.add(book);
    }
    
    @Override

        members.add(member);
    }
