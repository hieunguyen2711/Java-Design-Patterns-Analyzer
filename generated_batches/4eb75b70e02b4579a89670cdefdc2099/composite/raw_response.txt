### FILE: StockItem.java
```java
public abstract class StockItem {
    protected String name;
    protected int quantity;
    
    public StockItem(String name, int quantity) {
        this.name = name;
        this.quantity = quantity;
    }
    
    public abstract void add(StockItem item);
    public abstract void remove(StockItem item);
    public abstract int getQuantity();
    public abstract void display(int depth);
}
```

    }

        // Products cannot contain other items
        throw new UnsupportedOperationException("Products cannot contain other items");
    }
    
    @Override
    public int getQuantity() {
        return quantity;
    }
    
    @Override
    public void display(int depth) {
        StringBuilder indent = new StringBuilder();
        for (int i = 0; i < depth; i++) {
            indent.append("  ");
        }
        System.out.println(indent + "Product: " + name + " (SKU: " + sku + ") - Quantity: " + quantity);
    }
}
