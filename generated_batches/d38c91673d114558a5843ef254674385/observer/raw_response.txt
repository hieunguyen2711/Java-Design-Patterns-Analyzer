### FILE: Ticket.java
```java
import java.util.ArrayList;
import java.util.List;

public class Ticket {
    private String id;
    private String title;
    private String description;
    private Priority priority;
    private Status status;
    private List<Observer> observers;

    public Ticket(String id, String title, String description, Priority priority) {
        this.id = id;
        this.title = title;
        this.description = description;
        this.priority = priority;
        this.status = Status.OPEN;
        this.observers = new ArrayList<>();
    }

    public void addObserver(Observer observer) {
        observers.add(observer);
    }

    public void removeObserver(Observer observer) {
        observers.remove(observer);
    }

    public void setStatus(Status status) {
        this.status = status;
        notifyObservers();
    }

    public void assignTo(String assignee) {
        // In a real system, this would update the ticket's assignee
        System.out.println("Ticket " + id + " assigned to: " + assignee);
    }

    private void notifyObservers() {
        for (Observer observer : observers) {
            observer.update(this);
        }
    }

    // Getters
    public String getId() { return id; }
    public String getTitle() { return title; }
    public String getDescription() { return description; }
    public Priority getPriority() { return priority; }
    public Status getStatus() { return status; }
}
```

    public void update(Ticket ticket) {
        System.out.println("NOTIFICATION: Ticket " + ticket.getId() + 
                          " status changed to " + ticket.getStatus());
    }
}
