### FILE: TicketBookingService.java
```java
public interface TicketBookingService {
    void bookTicket(String eventId, String userId);
}
```

### FILE: ConcertTicketService.java
```java
public class ConcertTicketService implements TicketBookingService {
    @Override
    public void bookTicket(String eventId, String userId) {
        System.out.println("Booking concert ticket for event: " + eventId + " user: " + userId);
    }
}
```

### FILE: ServiceLocator.java
```java
import java.util.HashMap;
import java.util.Map;

public class ServiceLocator {
    private static Map<String, TicketBookingService> services = new HashMap<>();
    
    static {
        services.put("CONCERT", new ConcertTicketService());
        services.put("MOVIE", new MovieTicketService());
    }
    
    public static TicketBookingService getService(String serviceType) {
        return services.get(serviceType);
    }
}
```

### FILE: BookingController.java
```java
public class BookingController {
    public void processBooking(String eventType, String eventId, String userId) {
        TicketBookingService service = ServiceLocator.getService(eventType.toUpperCase());
        if (service != null) {
            service.bookTicket(eventId, userId);
        } else {
            System.out.println("No service found for type: " + eventType);
        }
    }
}
```

### FILE: Main.java
```java
public class Main {
    public static void main(String[] args) {
        BookingController controller = new BookingController();
        
        controller.processBooking("CONCERT", "C001", "user123");
        controller.processBooking("MOVIE", "M001", "user456");
        controller.processBooking("THEATER", "T001", "user789");
    }
}
```
