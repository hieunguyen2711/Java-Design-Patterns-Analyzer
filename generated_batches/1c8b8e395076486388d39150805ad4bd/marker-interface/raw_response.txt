```java
}
```

### FILE: GoldMembership.java
```java
package com.membership;

public class GoldMembership implements Membership {
    private final String memberId;
    private final String name;
    
    public GoldMembership(String memberId, String name) {
        this.memberId = memberId;
        this.name = name;
    }
    
    @Override
    public String getMemberId() {
        return memberId;
    }
    
    @Override
    public String getName() {
        return name;
    }
    
    @Override
    public boolean isActive() {
        return true;
    }
}
```

### FILE: MembershipService.java
```java
package com.membership;

import java.util.ArrayList;
import java.util.List;

public class MembershipService {
    private final List<Membership> memberships = new ArrayList<>();
    
    public void addMembership(Membership membership) {
        memberships.add(membership);
    }
    
    public boolean isGoldMember(Membership membership) {
        return membership instanceof GoldMembership;
    }
    
    public List<Membership> getAllMemberships() {
        return new ArrayList<>(memberships);
    }
}
```
