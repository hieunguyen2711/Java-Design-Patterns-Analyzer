### FILE: CheckInCommand.java

```java
package com.hotel.command;

import com.hotel.Room;
import com.hotel.Guest;

public class CheckInCommand implements Command {
    private Room room;
    private Guest guest;
    
    public CheckInCommand(Room room, Guest guest) {
        this.room = room;
        this.guest = guest;
    }
    
    @Override
    public void execute() {
        room.checkIn(guest);
    }
    
    @Override
    public void undo() {
        room.checkOut();
    }
}
```

### FILE: HotelCommandProcessor.java
```java
package com.hotel.command;

import java.util.Stack;

public class HotelCommandProcessor {
    private Stack<Command> commandHistory;
    
    public HotelCommandProcessor() {
        this.commandHistory = new Stack<>();
    }
    
    public void executeCommand(Command command) {
        command.execute();
        commandHistory.push(command);
    }
    
    public void undoLastCommand() {
        if (!commandHistory.isEmpty()) {
            Command lastCommand = commandHistory.pop();
            lastCommand.undo();
        }
    }
}
```
