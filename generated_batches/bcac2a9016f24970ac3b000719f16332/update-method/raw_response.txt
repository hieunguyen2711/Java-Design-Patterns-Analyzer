```java
package hotel;

public class Room {
    private int roomId;
    private String type;
    private double basePrice;
    
    public Room(int roomId, String type, double basePrice) {
        this.roomId = roomId;
        this.type = type;
        this.basePrice = basePrice;
    }
    
    public void updateBasePrice(double newPrice) {
        this.basePrice = newPrice;
    }
    
    public int getRoomId() {
        return roomId;
    }
    
    public String getType() {
        return type;
    }
    
    public double getBasePrice() {
        return basePrice;
    }
}
```

### FILE: RoomInventory.java
```java
package hotel;

import java.util.ArrayList;
import java.util.List;

public class RoomInventory {
    private List<Room> rooms;
    
    public RoomInventory() {
        this.rooms = new ArrayList<>();
    }
    
    public void addRoom(Room room) {
        rooms.add(room);
    }
    
    public void updateRoomPrice(int roomId, double newPrice) {
        for (Room room : rooms) {
            if (room.getRoomId() == roomId) {
                room.updateBasePrice(newPrice);
                break;
            }
        }
    }
    
    public List<Room> getRooms() {
        return new ArrayList<>(rooms);
    }
}
```
