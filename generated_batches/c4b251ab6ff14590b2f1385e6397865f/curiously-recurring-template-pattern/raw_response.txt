```java
package com.restaurant.order;

public abstract class Order<T extends Order<T>> {
    protected String orderId;
    protected double totalAmount;
    
    public Order(String orderId, double totalAmount) {
        this.orderId = orderId;
        this.totalAmount = totalAmount;
    }
    
    public abstract T withTotalAmount(double amount);
    
    public String getOrderId() {
        return orderId;
    }
    
    public double getTotalAmount() {
        return totalAmount;
    }
}
```

### FILE: FoodOrder.java

```java
package com.restaurant.order;

public class FoodOrder extends Order<FoodOrder> {
    private boolean isDelivery;
    
    public FoodOrder(String orderId, double totalAmount) {
        super(orderId, totalAmount);
        this.isDelivery = false;
    }
    
    @Override
    public FoodOrder withTotalAmount(double amount) {
        this.totalAmount = amount;
        return this;
    }
    
    public boolean isDelivery() {
        return isDelivery;
    }
    
    public void setDelivery(boolean delivery) {
        isDelivery = delivery;
    }
}
```

```java
package com.restaurant.order;

public class OrderProcessor {
    public static void processOrder(Order<?> order) {
        System.out.println("Processing order " + order.getOrderId() + 
                          " with amount $" + order.getTotalAmount());
    }
    
    public static void main(String[] args) {
        FoodOrder foodOrder = new FoodOrder("FO-001", 45.99);
        BeverageOrder beverageOrder = new BeverageOrder("BO-001", 12.50);
        
        processOrder(foodOrder);
        processOrder(beverageOrder);
    }
}
```
