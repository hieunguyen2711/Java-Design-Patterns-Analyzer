public class MenuItem {

        this.name = name;
        this.price = price;
        this.isDirty = false;
    }

        return name;
    }

        markAsDirty();
    }

        return price;
    }
    
    public void setPrice(double price) {
        this.price = price;
        markAsDirty();
    }
    
    public boolean isDirty() {
        return isDirty;
    }
    
    public void markAsDirty() {
        this.isDirty = true;
    }
    
    public void clearDirtyFlag() {
        this.isDirty = false;
    }
}
