### FILE: MenuItem.java

public class MenuItem {
    private String name;
    private double price;
    
    public MenuItem(String name, double price) {
        this.name = name;
        this.price = price;
    }
    
    public String getName() {
        return name;
    }
    
    public double getPrice() {
        return price;
    }
}
```

public interface OrderItemVisitor {
    void visit(MenuItem menuItem);
    void visit(Beverage beverage);
    void visit(Food food);
}
```

### FILE: PriceCalculationVisitor.java
```java
public class PriceCalculationVisitor implements OrderItemVisitor {
    private double totalPrice = 0.0;
    
    @Override
    public void visit(MenuItem menuItem) {
        totalPrice += menuItem.getPrice();
    }
    
    @Override
    public void visit(Beverage beverage) {
        totalPrice += beverage.getPrice();
    }
    
    @Override
    public void visit(Food food) {
        totalPrice += food.getPrice();
    }
    
    public double getTotalPrice() {
        return totalPrice;
    }
}
```
