### FILE: Message.java

```java
public abstract class Message {
    private final String sender;
    
    public Message(String sender) {
        this.sender = sender;
    }
    
    public String getSender() {
        return sender;
    }
    
    public abstract void process();
}
```

```java
    private final String accountId;

        super(sender);
        this.accountId = accountId;
        this.amount = amount;
```
